# Program 5: Comparing Complexities

## Aim

To demonstrate common time complexities—constant, logarithmic, linear, linearithmic, quadratic, and exponential—using simple Python examples.

## Program

```python
def constant(n):
    print("Constant: O(1)", n)

def logarithmic(n):
    while n > 1:
        n //= 2
    print("Logarithmic: O(log n)")

def linear(n):
    for i in range(n):
        pass
    print("Linear: O(n)")

def linearithmic(n):
    for i in range(n):
        j = n
        while j > 1:
            j //= 2
    print("Linearithmic: O(n log n)")

def quadratic(n):
    for i in range(n):
        for j in range(n):
            pass
    print("Quadratic: O(n^2)")

def exponential(n):
    def fib(k):
        if k <= 1:
            return k
        return fib(k - 1) + fib(k - 2)

    fib(n)
    print("Exponential example: O(2^n)")

n = 10
constant(n)
logarithmic(n)
linear(n)
linearithmic(n)
quadratic(n)
exponential(n)

```

```python
key = int(input("Enter element to search: "))

low = 0
high = len(arr) - 1

while low <= high:
    mid = (low + high) // 2

    if arr[mid] == key:
        print("Element found at index:", mid)

    elif arr[mid] < key:
```

# Program 11: Quick Sort

## Aim

To write a Python program to implement the quick sort algorithm for sorting array elements by selecting a pivot and partitioning the array around it.

## Program

```python
def quick_sort(arr):
    if len(arr) <= 1:
        return arr

    pivot = arr[len(arr) // 2]

    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]

    return quick_sort(left) + middle + quick_sort(right)

arr = list(map(int, input("Enter array elements: ").split()))

print("Sorted array:", quick_sort(arr))
```

# Program 12:  Merge Sort
## Aim

To write a Python program to implement merge sort using the divide-and-conquer technique to divide an array into smaller subarrays, sort them, and merge them into a sorted array.

## Program

```python
def merge_sort(arr):
    if len(arr) <= 1:
        return arr

    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])

    result = []
    i = j = 0

    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1

    result.extend(left[i:])
    result.extend(right[j:])

    return result

arr = list(map(int, input("Enter array elements: ").split()))

print("Sorted array:", merge_sort(arr))
```

```python
arr = list(map(int, input("Enter array elements: ").split()))

for i in range(1, len(arr)):
```

# Program 14: Bucket Sort
## Aim

To write a Python program to implement bucket sort by distributing elements into buckets, sorting individual buckets, and combining them to obtain the sorted array.

## Program

```python
def bucket_sort(arr):
    if not arr:
        return arr

    minimum = min(arr)
    maximum = max(arr)

    if minimum == maximum:
        return arr

    n = len(arr)
    buckets = [[] for _ in range(n)]

    for num in arr:
        index = int((num - minimum) / (maximum - minimum) * (n - 1))
        buckets[index].append(num)

    result = []

    for bucket in buckets:
        bucket.sort()
        result.extend(bucket)

    return result

arr = list(map(float, input("Enter numbers: ").split()))

print("Sorted array:", bucket_sort(arr))
```

# Program 28: Dijkstra's Algorithm

## Aim

## Program

```python
import heapq

def dijkstra(graph, source):
    distance = {vertex: float('inf') for vertex in graph}
    distance[source] = 0

    pq = [(0, source)]

    while pq:
        dist, vertex = heapq.heappop(pq)

        if dist > distance[vertex]:
            continue

        for neighbor, weight in graph[vertex]:
            new_dist = dist + weight

            if new_dist < distance[neighbor]:
                distance[neighbor] = new_dist
                heapq.heappush(pq, (new_dist, neighbor))

    print("Shortest distances from", source)
    for vertex, dist in distance.items():
        print(vertex, ":", dist)


graph = {
```

# Program 30: Comparison of Bellman–Ford and Dijkstra’s Algorithms Using Graph Representation

## Aim

To implement and compare Dijkstra’s and Bellman–Ford algorithms for finding the shortest paths from a source vertex in a weighted graph.

## Program

```python
import heapq

# Graph represented as an edge list
V = 5
edges = [
    (0, 1, 4),
    (0, 2, 1),
    (2, 1, 2),
    (1, 3, 1),
    (2, 3, 5),
    (3, 4, 3)
]


def dijkstra(V, edges, source):
    graph = [[] for _ in range(V)]

    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))

    dist = [float("inf")] * V
    dist[source] = 0
    pq = [(0, source)]

    while pq:
        d, u = heapq.heappop(pq)

        if d != dist[u]:
            continue

        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))

    return dist


def bellman_ford(V, edges, source):
    dist = [float("inf")] * V
    dist[source] = 0

    for _ in range(V - 1):
        changed = False

        for u, v, w in edges:
            if dist[u] != float("inf"):
                if dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w
                    changed = True

            # Graph is undirected
            if dist[v] != float("inf"):
                if dist[v] + w < dist[u]:
                    dist[u] = dist[v] + w
                    changed = True

        if not changed:
            break

    return dist


source = 0

print("Dijkstra's Algorithm:", dijkstra(V, edges, source))
print("Bellman-Ford Algorithm:",
      bellman_ford(V, edges, source))
```

# Program 31: Comparison of Prim’s and Kruskal’s Algorithms Using Graph Representation

## Aim

To implement and compare Prim’s and Kruskal’s algorithms for finding the Minimum Spanning Tree of a connected, weighted, undirected graph.

## Program

```python
import heapq

V = 5
edges = [
    (0, 1, 2),
    (0, 3, 6),
    (1, 2, 3),
    (1, 3, 8),
    (1, 4, 5),
    (2, 4, 7),
    (3, 4, 9)
]


def prim(V, edges):
    graph = [[] for _ in range(V)]

    for u, v, w in edges:
        graph[u].append((w, v))
        graph[v].append((w, u))

    visited = [False] * V
    pq = [(0, 0, -1)]
    mst = []
    total = 0

    while pq and len(mst) < V - 1:
        w, u, parent = heapq.heappop(pq)

        if visited[u]:
            continue

        visited[u] = True

        if parent != -1:
            mst.append((parent, u, w))
            total += w

        for weight, v in graph[u]:
            if not visited[v]:
                heapq.heappush(pq, (weight, v, u))

    return mst, total


def kruskal(V, edges):
    parent = list(range(V))
    rank = [0] * V

    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    def union(x, y):
        rx = find(x)
        ry = find(y)

        if rx == ry:
            return False

        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1

        return True

    mst = []
    total = 0

    for u, v, w in sorted(edges, key=lambda edge: edge[2]):
        if union(u, v):
            mst.append((u, v, w))
            total += w

            if len(mst) == V - 1:
                break

    return mst, total


for name, result in [
    ("Prim's", prim(V, edges)),
    ("Kruskal's", kruskal(V, edges))
]:
    mst, total = result
    print("\n", name, "Algorithm")
    print("Selected edges:", mst)
    print("Total MST weight:", total)
```

```python
    if n <= 1:
        return n

    dp = [0] * (n + 1)
    dp[1] = 1

    for i in range(2, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]

    return dp[n]


# Compare theoretical time complexities
n_values = list(range(1, 21))

recursive_complexity = [2 ** n for n in n_values]
dp_complexity = n_values

# Plot the graph
plt.plot(n_values, recursive_complexity,
         marker='o', label='Recursion O(2^n)')

plt.plot(n_values, dp_complexity,
         marker='s', label='Dynamic Programming O(n)')

plt.xlabel("Input Size (n)")
plt.ylabel("Approximate Number of Operations")
plt.title("Fibonacci: Recursion vs Dynamic Programming")
plt.legend()
plt.grid(True)

plt.show()
```

# Program 39: 0/1 Knapsack Problem Using Greedy Algorithm

## Aim

To implement the Greedy Algorithm in Python to solve the 0/1 Knapsack Problem by selecting complete items based on their profit-to-weight ratio without exceeding the maximum capacity of the knapsack.

## Program

```python

def greedy_knapsack(weights, profits, capacity):

    n = len(weights)

    # Calculate profit/weight ratio
    items = []

    for i in range(n):
        ratio = profits[i] / weights[i]
        items.append((ratio, weights[i], profits[i], i+1))

    # Sort by decreasing ratio
    items.sort(reverse=True)

    total_profit = 0
    total_weight = 0

    print("\nSelected Items:")

    for ratio, weight, profit, item_no in items:

        if total_weight + weight <= capacity:

            total_weight += weight
            total_profit += profit

            print("Item:", item_no,
                  "Weight:", weight,
                  "Profit:", profit)

    print("\nTotal Weight:", total_weight)
    print("Total Profit:", total_profit)

weights = [10, 20, 30]
```

# Program 40: Fractional Knapsack Problem Using Greedy Algorithm

## Aim

To implement the Greedy Algorithm in Python to solve the Fractional Knapsack Problem by selecting complete or fractional parts of items based on their profit-to-weight ratio to maximize the total profit within the given capacity.

## Program

```python

def fractional_knapsack(weights, profits, capacity):

    n = len(weights)

    # Calculate profit/weight ratio
    items = []

    for i in range(n):
        ratio = profits[i] / weights[i]
        items.append((ratio, weights[i], profits[i], i+1))

    # Sort items by decreasing ratio
    items.sort(reverse=True)

    total_profit = 0
    total_weight = 0

    print("\nSelected Items:")

    for ratio, weight, profit, item_no in items:

        if capacity >= weight:

            # Take complete item
            capacity -= weight
            total_weight += weight
            total_profit += profit

            print("Item:", item_no,
                  "Fraction: 1",
                  "Profit:", profit)

        else:

            # Take fractional item
            fraction = capacity / weight

            total_profit += profit * fraction
            total_weight += capacity

            print("Item:", item_no,
                  "Fraction:", round(fraction, 2),
                  "Profit:", round(profit * fraction, 2))

            capacity = 0
            break

    print("\nTotal Weight:", total_weight)
    print("Total Profit:", round(total_profit, 2))


# Main Program
n = int(input("Enter number of items: "))

weights = []
profits = []

for i in range(n):
    weights.append(float(input("Enter weight: ")))
    profits.append(float(input("Enter profit: ")))

capacity = float(input("Enter knapsack capacity: "))

fractional_knapsack(weights, profits, capacity)


```
